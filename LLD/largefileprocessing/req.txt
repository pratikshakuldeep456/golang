type ProcessorFactory interface {
    CreateReader(fileType FileType) ChunkReader
    CreateProcessor(fileType FileType) ChunkProcessor
    CreateWriter(fileType FileType) ChunkWriter
}

func CreateReader(fileType FileType) ChunkReader {
    switch fileType {
    case CSV:
        return NewCSVReader()
    case EXCEL:
        return NewExcelReader()
    case PDF:
        return NewPDFReader()
    default:
        return nil
    }
}
