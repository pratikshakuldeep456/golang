Factory = which strategy object do I hand back,
    given a ride type — this is the same relationship 
    you already built in parking lot between PaymentStrategyFactory
     and PaymentStrategy. You don't want if rideType == AUTO { ... } 
     else if rideType == CAB { ... } scattered across 
     your codebase every time you need a pricing strategy — 
     centralize that selection logic in one factory:

type RideType string
const (
    AUTO    RideType = "auto"
    CAB     RideType = "cab"
    PREMIUM RideType = "premium"
)
func GetPricingStrategy(rideType RideType) PricingStrategy {
    switch rideType {
    case AUTO:
        return &AutoPricingStrategy{}
    case CAB:
        return &CabPricingStrategy{}
    case PREMIUM:
        return &PremiumPricingStrategy{}
    default:
        return &CabPricingStrategy{} 
    }
}
